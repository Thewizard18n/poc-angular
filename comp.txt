// src/app/products/feature-product-list/product-list-page.component.ts
import {
  Component,
  ChangeDetectionStrategy,
  inject,
  computed
} from '@angular/core';
import { toSignal } from '@angular/core/rxjs-interop';
import { Router } from '@angular/router';
import { ProductListUseCase } from './product-list.usecase';
import { ProductRepository, ProductRepositoryMock, ProductModel } from '../data-access';
import { ProductCardComponent } from '../ui-product-card';

/**
 * Página de listagem de produtos.
 * Usa toSignal para converter Observable em Signal.
 */
@Component({
  selector: 'app-product-list-page',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [ProductCardComponent],
  providers: [
    ProductListUseCase,
    { provide: ProductRepository, useClass: ProductRepositoryMock }
  ],
  template: `
    <div class="page">
      <h1 class="page-title">Produtos</h1>

      @if (loading()) {
        <p class="loading">Carregando produtos...</p>
      } @else {
        <p class="total">{{ total() }} produtos encontrados</p>
        <div class="grid">
          @for (product of products(); track product.id) {
            <ui-product-card
              [product]="product"
              (selected)="onSelect($event)" />
          }
        </div>
      }
    </div>
  `,
  styles: [`
    .page { max-width: 1200px; }
    .page-title {
      font-size: 24px;
      color: var(--color-primary);
      margin-bottom: var(--spacing-md);
    }
    .total {
      color: var(--color-text-secondary);
      font-size: var(--font-size-sm);
      margin-bottom: var(--spacing-md);
    }
    .loading {
      color: var(--color-text-secondary);
    }
    .grid {
      display: grid;
      grid-template-columns: repeat(auto-fill, minmax(280px, 1fr));
      gap: var(--spacing-md);
    }
  `]
})
export class ProductListPageComponent {
  private useCase = inject(ProductListUseCase);
  private router = inject(Router);

  // toSignal — Observable → Signal
  // sem subscribe, sem async pipe, sem memory leak
  products = toSignal(this.useCase.execute(), { initialValue: [] });

  // computed deriva do Signal automaticamente
  total = computed(() => this.products().length);
  loading = computed(() => this.products().length === 0);

  onSelect(product: ProductModel) {
    this.router.navigate(['/products/detail', product.id]);
  }
}
